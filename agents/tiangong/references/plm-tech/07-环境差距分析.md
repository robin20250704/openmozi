> 检查时间：2026-09-02
> 检查对象：WSL Ubuntu-22.04（默认发行版，运行中）

## 3. 补齐操作脚本

### 3.1 P0 紧急补齐（阻断前端+后端+RAG 开发）

```bash
# === 在 WSL Ubuntu-22.04 中执行 ===

# G1: 安装 NVM + Node.js 22 LTS + pnpm
curl -o- https://raw.githubusercontent.com/nvm-sh/nvm/v0.40.0/install.sh | bash
source ~/.nvm/nvm.sh
nvm install 22
nvm use 22
npm install -g pnpm

# G2+G12: 安装 JDK 21 + 设置 JAVA_HOME
apt update && apt install -y openjdk-21-jdk
# 或用 SDKMAN:
# curl -s "https://get.sdkman.io" | bash && source ~/.sdkman/bin/sdkman-init.sh && sdk install java 21-tem
echo 'export JAVA_HOME=$(readlink -f $(which java) | sed "s:/bin/java::")' >> ~/.bashrc
source ~/.bashrc

# G3: 在 pdm-pg 安装 pgvector
docker exec pdm-pg apk add --no-cache postgresql16-pgvector 2>/dev/null || \
  docker exec pdm-pg sh -c 'cd /tmp && git clone https://github.com/pgvector/pgvector.git && cd pgvector && make && make install'
docker exec pdm-pg psql -U pdm -d pdm -c "CREATE EXTENSION IF NOT EXISTS vector;"
# 验证
docker exec pdm-pg psql -U pdm -d pdm -c "SELECT * FROM pg_extension WHERE extname='vector';"
```

### 3.2 P1 基础设施补齐（Kafka + Keycloak + ES 8.13）

```yaml
# docs/plm-tech/docker/dev-infra.yml
# 在 WSL 中: docker compose -f dev-infra.yml up -d

version: "3.9"

services:
  # G4: Kafka 3.7 KRaft 模式（去 ZooKeeper）
  kafka:
    image: confluentinc/cp-kafka:7.7.0
    container_name: plm-kafka
    ports:
      - "9092:9092"
    environment:
      KAFKA_NODE_ID: 1
      KAFKA_PROCESS_ROLES: "broker,controller"
      KAFKA_LISTENERS: "PLAINTEXT://:9092,CONTROLLER://:9093"
      KAFKA_ADVERTISED_LISTENERS: "PLAINTEXT://plm-kafka:9092"
      KAFKA_CONTROLLER_LISTENER_NAMES: "CONTROLLER"
      KAFKA_CONTROLLER_QUORUM_VOTERS: "1@plm-kafka:9093"
      KAFKA_INTER_BROKER_LISTENER_NAME: "PLAINTEXT"
      KAFKA_OFFSETS_TOPIC_REPLICATION_FACTOR: 1
      KAFKA_LOG_DIRS: /tmp/kraft-combined-logs
    networks: [plm_net]

  # G5: Keycloak 25
  keycloak:
    image: quay.io/keycloak/keycloak:25.0
    container_name: plm-keycloak
    ports:
      - "8443:8443"
      - "8090:8090"
    environment:
      KEYCLOAK_ADMIN: admin
      KEYCLOAK_ADMIN_PASSWORD: admin
      KC_HTTP_ENABLED: "true"
      KC_HTTP_PORT: "8090"
      KC_HOSTNAME: localhost
    command: ["start-dev"]
    networks: [plm_net]

  # G6: Elasticsearch 8.13（PLM 专用，端口 9201）
  es-plm:
    image: docker.elastic.co/elasticsearch/elasticsearch:8.13.4
    container_name: plm-es
    ports:
      - "9201:9200"
      - "9301:9300"
    environment:
      discovery.type: single-node
      xpack.security.enabled: "false"
      ES_JAVA_OPTS: "-Xms512m -Xmx512m"
    networks: [plm_net]

  # G7: Debezium Connect（CDC: PG→Kafka）
  debezium:
    image: quay.io/debezium/connect:2.7
    container_name: plm-debezium
    ports:
      - "8083:8083"
    environment:
      BOOTSTRAP_SERVERS: plm-kafka:9092
      GROUP_ID: plm-debezium
      CONFIG_STORAGE_TOPIC: debezium_configs
      OFFSET_STORAGE_TOPIC: debezium_offsets
      STATUS_STORAGE_TOPIC: debezium_statuses
    depends_on: [kafka]
    networks: [plm_net]

networks:
  plm_net:
    driver: bridge
```
